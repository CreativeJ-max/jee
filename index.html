import streamlit as st from google import genai from google.genai import types import json st.set_page_config(page_title="JEE AI Paper Analyzer", layout="wide") st.title("⚡ JEE AI Mock Paper & Screenshot Analyzer") st.markdown("Upload your question paper or screenshots and your worked rough sheets. The AI will solve the paper, grade your work, and analyze your weak subtopics.") # Sidebar for API Key and Uploads with st.sidebar: st.header("🔑 Configuration") api_key = st.text_input("Gemini API Key", type="password") st.header("📁 Upload Files") q_files = st.file_uploader("1. Question Paper / Screenshots", type=["pdf", "png", "jpg", "jpeg"], accept_multiple_files=True) w_files = st.file_uploader("2. Your Rough Work / Solutions", type=["pdf", "png", "jpg", "jpeg", "csv"], accept_multiple_files=True) run_btn = st.button("🚀 Analyze Paper", type="primary", use_container_width=True) if run_btn: if not api_key: st.error("Please enter your Google Gemini API Key.") elif not q_files: st.error("Please upload at least one question paper or screenshot.") else: with st.spinner("🤖 AI is scanning your files, solving the questions, and evaluating your mistakes..."): try: client = genai.Client(api_key=api_key) # Upload files to Gemini uploaded_g_files = [] for f in q_files + w_files: g_file = client.files.upload(file=f) uploaded_g_files.append(g_file) prompt = """ You are an expert JEE coach. Analyze these question files and student work files. Provide a JSON response with: 1. "summary": estimated total score (out of 300), accuracy percentage, and a list of weak subtopics. 2. "questions": an array of analyzed questions containing q_no, subject, subtopic, status (Correct/Incorrect), user_ans, correct_ans, mistake_type, solution, and mistake_desc. Return strictly valid JSON. """ response = client.models.generate_content( model='gemini-2.5-flash', contents=[*uploaded_g_files, prompt], config=types.GenerateContentConfig(response_mime_type="application/json") ) data = json.loads(response.text) st.session_state['data'] = data st.success("Analysis Complete!") except Exception as e: st.error(f"Error during analysis: {e}") # Display Results if available if 'data' in st.session_state: data = st.session_state['data'] summary = data.get("summary", {}) col1, col2, col3 = st.columns(3) col1.metric("Estimated Score", summary.get("total_score", "N/A")) col2.metric("Accuracy", summary.get("accuracy", "N/A")) col3.metric("Weak Subtopics Found", len(summary.get("weak_topics", []))) st.subheader("Detailed Question Breakdown") for q in data.get("questions", []): with st.expander(f"Q{q.get('q_no')} | {q.get('subject')} - {q.get('subtopic')} ({q.get('status')})"): st.write(f"**Mistake Type:** {q.get('mistake_type')}") st.write(f"**Step-by-Step Solution:** {q.get('solution')}") st.write(f"**AI Root Cause:** {q.get('mistake_desc')}")
